# One line per case: DECISION RULE COMMAND. DECISION is DENY or ALLOW. RULE is
# the pack rule that must match (pack.id:pattern) for a DENY, or `-` for an
# ALLOW. scripts/test-dcg-packs runs each command through `dcg test
# --format json` from the repository root under .dcg/test-config.toml and
# compares decision and rule. Every safe pattern has a case that would match
# its destructive rule without it, so removing an exemption fails the run.

# rune.search
DENY  rune.search:grep-use-rg        grep -r pattern .
DENY  rune.search:find-use-fd        find . -name '*.md'
DENY  rune.search:grep-alt-forms     /usr/bin/grep pattern file
DENY  rune.search:grep-alt-forms     xargs grep pattern
DENY  rune.search:find-use-fd|rune.search:find-alt-forms  busybox find . -name x
DENY  rune.search:find-use-fd|rune.search:find-alt-forms  /usr/bin/find . -name x
ALLOW -                              grep --version
ALLOW -                              find --help
ALLOW -                              git grep pattern

ALLOW -                              sed -n '10,20p' README.md
